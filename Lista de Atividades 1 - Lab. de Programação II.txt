public class ListaDeExercicios1 {
    public static void main(String[] args) {
        
        double[]notas={10,5,3,8,9,6,10,10,9,5};
        double media;
        double maior=0;
        double menor=Integer.MAX_VALUE;
        int soma=0;
        int k=0,j=0;
        
        for(int i=0;i<10;i++){
            
            //media
            soma+=notas[i];
            
            //maior e menor
            if(notas[i]>maior){
                maior=notas[i];
            }
            
            if(notas[i]<menor){
                menor=notas[i];
            }
            
            //maior que 7
            if(notas[i]>7.0){
                k++;
            }  
        }
        media=soma/10;
        //maior que media
        for(int i=0;i<10;i++){
            if(notas[i]<media){
                j++;
            }
        }
        
        System.out.println("Media: "+media); 
        System.out.println("Maior: "+maior);
        System.out.println("Menor: "+menor);
        System.out.println("Maior que 7: "+k);
        System.out.println("Menor que a media: "+j);
    }
}

import java.util.Scanner;

public class DiagonalMatriz {
    public static void main(String[] args) {
        Scanner ler=new Scanner(System.in);
        
        int[][]matriz=new int[4][4];
        int somaDP=0;
        int somaDS=0;
        
        //Inserindo valores
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                System.out.println("Valor "+"["+(i+1)+"]["+(j+1)+"]");
                matriz[i][j]=ler.nextInt();
            }
        }
        System.out.println(" ");
        
        //Printando
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                System.out.print(matriz[i][j]+"  ");
            }
            System.out.println(" ");
        }
        System.out.println(" ");
        
        //Diagonal Principal
        System.out.println("Diagonal Principal");
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                if(i==j){
                    System.out.println(matriz[i][j]+" ");
                    somaDP+=matriz[i][j];
                }
            }
        }
        System.out.println("Soma da diagonal: "+somaDP);
        
        //Diagonal Secundaria
        System.out.println("Diagonal Secundaria");
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                if(j==3-i){
                    System.out.println(matriz[i][j]+" ");
                    somaDS+=matriz[i][j];
                }
            }
        }
        System.out.println("Soma da diagnoal: "+somaDS);
        
        //Verificando as somas
        if(somaDP>somaDS){
            System.out.println("Maior soma: Diagonal Principal");          
        }else if(somaDP==somaDS){
            System.out.println("As somas sao iguais");
        }else{
            System.out.println("Maior soma: Diagonal Secundaria");
        }
    }    
}

import java.util.Scanner;

public class ValoresRepetidos {
    public static void main(String[] args) {
        Scanner ler=new Scanner(System.in);
        
        int[]vetor=new int[15];
        int[]diferente=new int[15];
        int quant=0;
        
        //Inserindo valores
        for(int i=0;i<vetor.length;i++){
            System.out.println("Valor "+"["+(i+1)+"]");
            vetor[i]=ler.nextInt();
        }
        System.out.println(" ");
        
        //Printando Vetor
        System.out.println("VETOR: ");
        for(int i=0;i<vetor.length;i++){
            System.out.print(vetor[i]+" ");
        }
        System.out.println(" ");
        
        //Comparando os vetores
        for(int i=0;i<vetor.length;i++){
            int rep=0;
            for(int j=0;j<vetor.length;j++){
                if(vetor[i]==diferente[j]){
                    rep=1;
                }
            }
            if(rep==0){
                diferente[quant]=vetor[i];
                quant++;
            }
        }
        
        //Resultados
        System.out.println("Valores Diferentes: ");
        for(int i=0;i<quant;i++){
            System.out.print(diferente[i]+" ");
        }
        System.out.println(" ");
        
        System.out.println("Quantidade de valores diferentes: "+quant);
    }    
}

import java.util.Scanner;

public class MatrizRotacao {
    public static void main(String[] args) {
        Scanner ler=new Scanner(System.in);
        
        int[][]matriz=new int[5][5];
        int[][]rot=new int[5][5];
        
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                System.out.println("Valor "+"["+(i+1)+"]["+(j+1)+"]");
                matriz[i][j]=ler.nextInt();
            }
        }
        
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                rot[j][4-i]=matriz[i][j];
            }
        }
        
        System.out.println("Matriz Original");
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                System.out.print(matriz[i][j]+" ");
            }
            System.out.println(" ");
        }
        
        System.out.println("Matriz Rotacionada");
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                System.out.print(rot[i][j]+" ");
            }
            System.out.println(" ");
        }
    }
}

import java.util.Scanner;

public class BubbleSort {
    public static void main(String[] args) {
        Scanner ler=new Scanner(System.in);
        
        int[]vetor=new int[20];
        
        //Inserindo vetor
        System.out.println("INSERINDO VALORES");
        for(int i=0;i<vetor.length;i++){
            System.out.print("Valor ["+(i+1)+"]: ");
            vetor[i]=ler.nextInt();
        }
        System.out.println("");
        
        //Imprimindo vetor
        System.out.println("VETOR DIGITADO");
        for(int i=0;i<vetor.length;i++){
            System.out.print(vetor[i]+" ");
        }
        
        //BubbleSort
        int trocas=0;
        int comp=0;
        for(int i=0;i<vetor.length-1;i++){
            int changes=0;
            for(int j=0;j<vetor.length-1;j++){
                comp++;
                if(vetor[j]>vetor[j+1]){
                    int temp = vetor[j];
                    vetor[j] = vetor[j + 1];
                    vetor[j + 1] = temp;
                    changes++;
                    trocas++;
                }
            }
            if(changes==0)break;
        }
        System.out.println("\n");

        //Imprimindo o vetor organizado
        System.out.println("VETOR ORDENADO");
        for(int i=0;i<vetor.length;i++){
            System.out.print(vetor[i]+" ");
        }
        System.out.println("\n");
        
        //Estatisticas
        int mediana;
        mediana=(comp+trocas)/2;
        System.out.println("ESTATISTICAS");
        System.out.println("Comparacoes realizadas: "+comp);
        System.out.println("Trocas realizadas: "+trocas);
        System.out.println("Mediana: "+mediana);
    }    
}

import java.util.Scanner;

public class MatrizMagica {

    public static void main(String[] args) {
        Scanner ler=new Scanner(System.in);
        
        int[][]matriz=new int[3][3];
        boolean magico=true;
        
        //Inserindo os valores
        System.out.println("INSERINDO VALORES");
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                System.out.println("Valor ["+(i+1)+"]["+(j+1)+"]");
                matriz[i][j]=ler.nextInt();
            }
        }
        System.out.println("");
        
        //Imprimindo a matriz
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                System.out.print(matriz[i][j]+"  ");
            }
            System.out.println(" ");
        } 
        System.out.println("");
        
        //Soma de referênca para comparação
        int somaRef=0;
        for(int i=0;i<matriz.length;i++){
            somaRef+=matriz[0][i];
        }
        
        //Verificando a soma da linha
        int somaLinha=0;
        boolean Vlinha=true;
        for(int i=0;i<matriz.length;i++){
            somaLinha=0;
            for(int j=0;j<matriz.length;j++){
                somaLinha+=matriz[i][j];
            }
            if(somaLinha!=somaRef){
                magico=false;
                Vlinha=false;
            }
        }
        
        //Verificando a soma da coluna
        int somaColuna=0;
        boolean Vcoluna=true;
        for(int j=0;j<matriz.length;j++){
            somaColuna=0;
            for(int i=0;i<matriz.length;i++){
                somaColuna+=matriz[i][j];
            }
            if(somaColuna!=somaRef){
                magico=false;
                Vcoluna=false;
            }
        }
        
        //Comparando a diagonal principal
        int somaDP=0;
        boolean VDP=true;
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                if(i==j){
                    somaDP+=matriz[i][j];
                }
            }
        }
        if(somaDP!=somaRef){
            magico=false;
            VDP=false;
        }
        
        //Comparando a secundaria
        int somaDS=0;
        boolean VDS=true;
        for(int i=0;i<matriz.length;i++){
            for(int j=0;j<matriz.length;j++){
                if(j==2-i){
                    somaDS+=matriz[i][j];
                }
            }
        }
        if(somaDS!=somaRef){
            magico=false;
            VDS=false;
        }

        //Resultados
       if(magico==true){
           System.out.println("A matriz e magica.");
       }else{
           System.out.println("A matriz nao e magica.");
       
           //Criterios nao atendidos
           //Vlinha
           //Vcoluna
           //VDP
           //VDS

	   System.out.println("Criterios nao atendidos/n.");
           if(!Vlinha){
               System.out.println("-Soma da linha-");
           }
           if(!Vcoluna){
               System.out.println("-Soma da coluna-");
           }
           if(!VDP){
               System.out.println("-Soma da diagonal principal-");
           }
           if(!VDS){
               System.out.println("-Soma da diagonal secundaria-");
           }
       }
    }
